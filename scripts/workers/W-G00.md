### 2. scripts/test-yk-push.ps1 内容
```powershell
param([string]$Reason = "PM 测试")
$ymlPath = "E:\DEMO\数据采集\DataupLoad\config\application-prod.yml"
$logPath = "E:\DEMO\数据采集\DataupLoad\logs\test-yk-push.log"

Write-Host "[$(Get-Date)] 老板同意推了：$Reason" | Tee-Object $logPath -Append
Write-Host "[$(Get-Date)] 改 yk.enable = true" | Tee-Object $logPath -Append
(Get-Content $ymlPath) -replace 'enable: false', 'enable: true' | Set-Content $ymlPath

# 30 秒后自动回滚
$job = Start-Job -ScriptBlock {
    Start-Sleep -Seconds 30
    $yml = "E:\DEMO\数据采集\DataupLoad\config\application-prod.yml"
    (Get-Content $yml) -replace 'enable: true', 'enable: false' | Set-Content $yml
    Write-Host "[$(Get-Date)] 自动回滚 yk.enable = false（30 秒超时）" | Tee-Object "E:\DEMO\数据采集\DataupLoad\logs\test-yk-push.log" -Append
}

Write-Host "[$(-Date)] PM 现在可以推测试，30 秒后自动熔断"
Write-Host "Job ID: $($job.Id)"
Write-Host "等老板说'继续推'可以再跑此脚本"
```

fi
